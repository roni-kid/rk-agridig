# RK AgriDig — Python dependencies
# Target: Python >= 3.11, CPU-only, offline-first (Ubuntu 22.04 LTS dev target,
# WSL2 dev environment). Model inference itself runs through the local Ollama
# server / llama.cpp -- these packages cover the Gradio UI, the HTTP bridge to
# Ollama, and the Phase 2 offline evaluation pipeline (batch runner + scorer).
#
# Install with:
#   pip install -r requirements.txt --break-system-packages
# (--break-system-packages needed on Debian/Ubuntu system Python; omit if
# installing inside a venv/virtualenv, which is the recommended setup.)

# --- UI (app.py) ---
gradio>=4.44,<5.0

# --- Ollama HTTP bridge (src/ollama_client.py) ---
requests>=2.31,<3.0

# --- Phase 2 evaluation (src/evaluator.py, src/prompt_engine.py) ---
# sacrebleu: standardized, reproducible BLEU-4 scoring (not nltk -- see
# evaluator.py's module docstring for why).
sacrebleu>=2.4,<3.0

# sentence-transformers pulls in torch + transformers + numpy as transitive
# dependencies automatically -- pinned here only at the top level so the
# actually-used all-MiniLM-L6-v2 embedding model loads correctly.
#
# IMPORTANT (CPU-only target hardware): a bare `pip install sentence-transformers`
# on Linux will pull the CUDA-enabled torch build by default, which drags in
# several GB of unused NVIDIA wheels on this AMD/CPU-only machine. Install the
# CPU-only torch wheel FIRST, then this file, e.g.:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu --break-system-packages
#   pip install -r requirements.txt --break-system-packages
sentence-transformers>=3.0,<4.0

# scikit-learn: only used for cosine_similarity in evaluator.py's semantic
# similarity metric -- a full ML framework is overkill for one function, but
# it's the standard, well-tested implementation and already a transitive dep
# of several other packages in this list.
scikit-learn>=1.4,<2.0

# --- Phase 3 additions (image analysis + voice output) ---
# Added when src/image_symptom_extractor.py and src/voice_output.py were
# introduced. The original requirements.txt predated these modules.

# opencv-python: classical CV heuristics for leaf photo analysis
# (color segmentation, lesion detection). Chosen over a vision-capable LLM
# because Phi-3-mini-4k-instruct has no vision tower, and any model that
# does would blow past the 7GB RAM ceiling. This is deterministic array math,
# not a neural forward pass -- negligible RAM and thermal impact.
opencv-python>=4.9,<5.0

# numpy: explicit pin even though sentence-transformers already pulls it in
# transitively. app.py and image_symptom_extractor.py both use np directly,
# so the dependency should be declared, not assumed.
numpy>=1.26,<3.0

# pyttsx3: offline text-to-speech wrapping the OS's native engine (espeak-ng
# on WSL2/Linux). Near-zero RAM/CPU cost compared to neural TTS, which would
# compete with Phi-3-mini for the thermal and memory budget.
# NOTE: also requires the espeak-ng system package:
#   sudo apt-get install espeak-ng
pyttsx3>=2.90

# --- Dataset + model download (scripts/, models/) ---
# huggingface-hub: used by models/download_model.py to pull the Phi-3-mini
# Q4_K_M GGUF from HuggingFace. Not a runtime dependency of the app itself.
huggingface-hub>=0.23,<1.0

# datasets: used by scripts/create_eval_dataset.py to load GhanaAgricVQA
# train/test splits. Also not a runtime dep -- only needed for eval prep.
datasets>=2.20,<4.0

# --- Benchmarking / thermal monitoring ---
# psutil: CPU temperature + peak RAM measurement in benchmarks/thermal_monitor.py.
# Also not a runtime dep -- only needed during profiling runs.
psutil>=5.9,<7.0

# --- Note on what's intentionally NOT listed here ---
# llama.cpp / llama-bench / Ollama itself: native binaries, not pip packages.

# kept out of this file since it's a competition submission/scoring tool, not
